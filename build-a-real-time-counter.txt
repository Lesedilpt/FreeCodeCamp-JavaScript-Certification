<body>

  <h2>Real Time Character Counter</h2>

  <textarea id="text-input" placeholder="Type something..."></textarea>
  <p id="char-count">Character Count: 0/50</p>

  <script>
    const textarea = document.getElementById('text-input');
    const charCount = document.getElementById('char-count');
    const maxLength = 50;

    textarea.addEventListener('input', () => {
      let text = textarea.value;

      // Trim text if it exceeds maxLength
      if (text.length > maxLength) {
        text = text.slice(0, maxLength);
        textarea.value = text; // Update the textarea value
      }

      // Update character count
      charCount.textContent = `Character Count: ${text.length}/50`;

      // Add or remove red color if max length reached
      if (text.length >= maxLength) {
        charCount.classList.add('max');
      } else {
        charCount.classList.remove('max');
      }
    });
  </script>

</body>
