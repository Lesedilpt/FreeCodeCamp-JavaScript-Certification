function smallestCommons(arr) {
  // Sort the array to get the lower and upper bounds
  const [min, max] = arr[0] < arr[1] ? arr : [arr[1], arr[0]];

  // Helper function to calculate GCD (Greatest Common Divisor)
  const gcd = (a, b) => b === 0 ? a : gcd(b, a % b);

  // Helper function to calculate LCM (Least Common Multiple)
  const lcm = (a, b) => (a * b) / gcd(a, b);

  // Create an array of numbers in the range [min, max]
  const range = [];
  for (let i = min; i <= max; i++) {
    range.push(i);
  }

  // Reduce the array to its LCM
  return range.reduce((acc, curr) => lcm(acc, curr));
}
