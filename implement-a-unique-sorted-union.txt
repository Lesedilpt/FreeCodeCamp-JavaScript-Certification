function uniteUnique(...arrays) {
  const result = [];

  // Loop through each array in the arguments
  arrays.forEach(arr => {
    arr.forEach(value => {
      // Add value only if it’s not already in result
      if (!result.includes(value)) {
        result.push(value);
      }
    });
  });

  return result;
}
