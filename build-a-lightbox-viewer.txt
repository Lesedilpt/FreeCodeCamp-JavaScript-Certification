  <!-- Gallery -->
  <div class="gallery">
    <img src="https://cdn.freecodecamp.org/curriculum/labs/stonehenge-thumbnail.jpg" class="gallery-item" alt="Stonehenge">
    <img src="https://cdn.freecodecamp.org/curriculum/labs/storm-thumbnail.jpg" class="gallery-item" alt="Storm">
    <img src="https://cdn.freecodecamp.org/curriculum/labs/trees-thumbnail.jpg" class="gallery-item" alt="Trees">
  </div>

  <!-- Lightbox -->
  <div class="lightbox">
    <button id="close-btn">&times;</button>
    <img id="lightbox-image" src="" alt="Full Size">
  </div>

  <script>
    const galleryItems = document.querySelectorAll('.gallery-item');
    const lightbox = document.querySelector('.lightbox');
    const lightboxImage = document.getElementById('lightbox-image');
    const closeBtn = document.getElementById('close-btn');

    // Function to open lightbox
    galleryItems.forEach(item => {
      item.addEventListener('click', () => {
        // Replace -thumbnail with full size image
        const fullSrc = item.src.replace('-thumbnail', '');
        lightboxImage.src = fullSrc;
        lightbox.style.display = 'flex';
      });
    });

    // Function to close lightbox
    closeBtn.addEventListener('click', () => {
      lightbox.style.display = 'none';
    });

    // Clicking outside image closes lightbox
    lightbox.addEventListener('click', (e) => {
      if (e.target === lightbox) {
        lightbox.style.display = 'none';
      }
    });
  </script>

</body>
