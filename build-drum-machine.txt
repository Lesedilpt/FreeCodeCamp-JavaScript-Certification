** start of script.js **

const display = document.getElementById('display');
const pads = document.querySelectorAll('.drum-pad');

// Play sound and update display
function playSound(e) {
  const audio = e.target.querySelector('audio') || document.getElementById(e.key.toUpperCase());
  if (!audio) return;

  audio.currentTime = 0;
  audio.play();

  const parent = audio.parentElement;
  display.innerText = parent.id;
}

// Click event
pads.forEach(pad => pad.addEventListener('click', playSound));

// Keyboard event
document.addEventListener('keydown', (e) => {
  const key = e.key.toUpperCase();
  const audio = document.getElementById(key);
  if (!audio) return;

  audio.currentTime = 0;
  audio.play();
  display.innerText = audio.parentElement.id;
});


** end of script.js **
