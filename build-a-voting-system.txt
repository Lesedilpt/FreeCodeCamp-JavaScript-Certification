// 1. Initialize poll as a new Map
const poll = new Map();

// 2. Function to add an option
function addOption(option) {
  if (!option || option.trim() === "") {
    return "Option cannot be empty.";
  }

  if (poll.has(option)) {
    return `Option "${option}" already exists.`;
  }

  poll.set(option, new Set());
  return `Option "${option}" added to the poll.`;
}

// 3. Function to vote for an option
function vote(option, voterId) {
  if (!poll.has(option)) {
    return `Option "${option}" does not exist.`;
  }

  const voters = poll.get(option);
  if (voters.has(voterId)) {
    return `Voter ${voterId} has already voted for "${option}".`;
  }

  voters.add(voterId);
  return `Voter ${voterId} voted for "${option}".`;
}

// 4. Function to display poll results
function displayResults() {
  let results = "Poll Results:\n";
  poll.forEach((voters, option) => {
    results += `${option}: ${voters.size} votes\n`;
  });
  return results.trim(); // Remove trailing newline
}

// Add at least 3 options
addOption("Turkey");
addOption("Morocco");
addOption("Egypt");

// Add at least 3 votes
vote("Turkey", "traveler1");
vote("Morocco", "traveler2");
vote("Egypt", "traveler3");

// Additional votes to test duplicate prevention
vote("Turkey", "traveler4");
vote("Morocco", "traveler5");
vote("Turkey", "traveler1"); // duplicate, should not count

// Test: display results
console.log(displayResults());
