function spinalCase(str) {
  // 1. Replace any underscore or space with a space for normalization
  let normalized = str.replace(/[_\s]+/g, ' ');

  // 2. Add a space before any uppercase letter that is not at the start
  normalized = normalized.replace(/([a-z])([A-Z])/g, '$1 $2');

  // 3. Convert to lowercase
  normalized = normalized.toLowerCase();

  // 4. Replace spaces with hyphens
  const spinal = normalized.replace(/\s+/g, '-');

  return spinal;
}

// Examples to test
console.log(spinalCase("This Is Spinal Tap"));        // "this-is-spinal-tap"
console.log(spinalCase("thisIsSpinalTap"));           // "this-is-spinal-tap"
console.log(spinalCase("The_Andy_Griffith_Show"));    // "the-andy-griffith-show"
console.log(spinalCase("Teletubbies say Eh-oh"));     // "teletubbies-say-eh-oh"
console.log(spinalCase("AllThe-small Things"));       // "all-the-small-things"
