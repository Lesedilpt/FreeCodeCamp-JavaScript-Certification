function sumAll(arr) {
  // Get the smaller and larger numbers
  const min = Math.min(arr[0], arr[1]);
  const max = Math.max(arr[0], arr[1]);
  
  let sum = 0;
  
  // Sum all numbers from min to max inclusive
  for (let i = min; i <= max; i++) {
    sum += i;
  }
  
  return sum;
}
