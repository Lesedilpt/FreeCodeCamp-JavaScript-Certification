function frankenSplice(arr1, arr2, index) {
  // Make a copy of arr2 so we don't modify it
  let newArr = arr2.slice();

  // Insert all elements of arr1 into newArr at the given index
  newArr.splice(index, 0, ...arr1);

  return newArr;
}
