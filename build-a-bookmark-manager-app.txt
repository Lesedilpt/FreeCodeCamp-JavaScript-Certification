// Get bookmarks from localStorage
function getBookmarks() {
  try {
    const bookmarks = JSON.parse(localStorage.getItem("bookmarks"));
    if (!Array.isArray(bookmarks)) return [];

    // Check that each item is an object with required keys
    const isValid = bookmarks.every(
      b => b && typeof b === "object" && "name" in b && "category" in b && "url" in b
    );

    return isValid ? bookmarks : [];
  } catch (error) {
    return [];
  }
}

// Toggle form visibility
function displayOrCloseForm() {
  document.getElementById("main-section").classList.toggle("hidden");
  document.getElementById("form-section").classList.toggle("hidden");
}

// Toggle category list visibility
function displayOrHideCategory() {
  document.getElementById("main-section").classList.toggle("hidden");
  document.getElementById("bookmark-list-section").classList.toggle("hidden");
}

// Add a new bookmark
function addBookmark() {
  const nameInput = document.getElementById("name");
  const urlInput = document.getElementById("url");
  const categoryDropdown = document.getElementById("category-dropdown");
  const category = categoryDropdown.value;

  if (!nameInput.value.trim()) {
    alert("Please provide a name");
    return;
  }

  const newBookmark = {
    name: nameInput.value.trim(),
    category: category,
    url: urlInput.value.trim()
  };

  const bookmarks = getBookmarks();
  bookmarks.push(newBookmark);
  localStorage.setItem("bookmarks", JSON.stringify(bookmarks));

  // Reset form
  nameInput.value = "";
  urlInput.value = "";

  displayOrCloseForm();
}

// View bookmarks by category
function viewCategory() {
  const categoryDropdown = document.getElementById("category-dropdown");
  const category = categoryDropdown.value;
  document.querySelector(".category-name").innerText = category;

  const bookmarks = getBookmarks();
  const categoryBookmarks = bookmarks.filter(b => b.category === category);

  const categoryList = document.getElementById("category-list");
  categoryList.innerHTML = ""; // Clear previous entries

  if (categoryBookmarks.length === 0) {
    categoryList.innerHTML = "<p>No Bookmarks Found</p>";
  } else {
    categoryBookmarks.forEach(b => {
      const safeId = b.name.replace(/\s+/g, "-").replace(/[^a-zA-Z0-9-_]/g, "");
      const radioHTML = `
        <input type="radio" id="${safeId}" name="bookmark" value="${b.name}">
        <label for="${safeId}"><a href="${b.url}" target="_blank">${b.name}</a></label>
      `;
      categoryList.innerHTML += radioHTML;
    });
  }

  displayOrHideCategory();
}

// Delete selected bookmark
function deleteBookmark() {
  const categoryDropdown = document.getElementById("category-dropdown");
  const category = categoryDropdown.value;

  const selectedRadio = document.querySelector('input[name="bookmark"]:checked');
  if (!selectedRadio) return;

  const selectedName = selectedRadio.value;

  let bookmarks = getBookmarks();
  bookmarks = bookmarks.filter(b => !(b.category === category && b.name === selectedName));
  localStorage.setItem("bookmarks", JSON.stringify(bookmarks));

  // Update list display
  viewCategory();
}

// Event listeners
document.getElementById("add-bookmark-button").addEventListener("click", () => {
  const categoryDropdown = document.getElementById("category-dropdown");
  document.querySelector(".category-name").innerText = categoryDropdown.value;
  displayOrCloseForm();
});

document.getElementById("close-form-button").addEventListener("click", displayOrCloseForm);

document.getElementById("add-bookmark-button-form").addEventListener("click", addBookmark);

document.getElementById("view-category-button").addEventListener("click", viewCategory);

document.getElementById("close-list-button").addEventListener("click", displayOrHideCategory);

document.getElementById("delete-bookmark-button").addEventListener("click", deleteBookmark);
