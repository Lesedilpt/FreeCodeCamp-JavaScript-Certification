** start of script.js **

// Select DOM elements
const form = document.getElementById('form');
const fullNameInput = document.getElementById('full-name');
const emailInput = document.getElementById('email');
const orderNoInput = document.getElementById('order-no');
const productCodeInput = document.getElementById('product-code');
const quantityInput = document.getElementById('quantity');
const complaintsGroup = document.getElementById('complaints-group');
const complaintDescription = document.getElementById('complaint-description');
const otherComplaintCheckbox = document.getElementById('other-complaint');
const solutionsGroup = document.getElementById('solutions-group');
const solutionDescription = document.getElementById('solution-description');
const otherSolutionRadio = document.getElementById('other-solution');
const messageBox = document.getElementById('message-box');

// Validation functions
function validateForm() {
    const validations = {
        "full-name": fullNameInput.value.trim() !== "",
        "email": /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailInput.value),
        "order-no": /^2024\d{6}$/.test(orderNoInput.value),
        "product-code": /^[a-zA-Z]{2}\d{2}-[a-zA-Z]\d{3}-[a-zA-Z]{2}\d$/.test(productCodeInput.value),
        "quantity": Number(quantityInput.value) > 0,
        "complaints-group": Array.from(complaintsGroup.querySelectorAll('input[type="checkbox"]')).some(cb => cb.checked),
        "complaint-description": !otherComplaintCheckbox.checked || complaintDescription.value.trim().length >= 20,
        "solutions-group": Array.from(solutionsGroup.querySelectorAll('input[type="radio"]')).some(rb => rb.checked),
        "solution-description": !otherSolutionRadio.checked || solutionDescription.value.trim().length >= 20
    };
    return validations;
}

function isValid(validations) {
    return Object.values(validations).every(val => val);
}

// Event handlers to set border color dynamically
function setBorderColor(element, isValid) {
    element.style.borderColor = isValid ? "green" : "red";
}

function setGroupBorderColor(group, isValid) {
    group.style.borderColor = isValid ? "green" : "red";
}

// Add change events for each input
fullNameInput.addEventListener('change', () => setBorderColor(fullNameInput, fullNameInput.value.trim() !== ""));
emailInput.addEventListener('change', () => setBorderColor(emailInput, /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailInput.value)));
orderNoInput.addEventListener('change', () => setBorderColor(orderNoInput, /^2024\d{6}$/.test(orderNoInput.value)));
productCodeInput.addEventListener('change', () => setBorderColor(productCodeInput, /^[a-zA-Z]{2}\d{2}-[a-zA-Z]\d{3}-[a-zA-Z]{2}\d$/.test(productCodeInput.value)));
quantityInput.addEventListener('change', () => setBorderColor(quantityInput, Number(quantityInput.value) > 0));

complaintsGroup.querySelectorAll('input[type="checkbox"]').forEach(cb => {
    cb.addEventListener('change', () => {
        setGroupBorderColor(complaintsGroup, Array.from(complaintsGroup.querySelectorAll('input[type="checkbox"]')).some(cb => cb.checked));
    });
});

complaintDescription.addEventListener('change', () => {
    if (otherComplaintCheckbox.checked) {
        setBorderColor(complaintDescription, complaintDescription.value.trim().length >= 20);
    } else {
        complaintDescription.style.borderColor = "";
    }
});

solutionsGroup.querySelectorAll('input[type="radio"]').forEach(rb => {
    rb.addEventListener('change', () => {
        setGroupBorderColor(solutionsGroup, Array.from(solutionsGroup.querySelectorAll('input[type="radio"]')).some(rb => rb.checked));
    });
});

solutionDescription.addEventListener('change', () => {
    if (otherSolutionRadio.checked) {
        setBorderColor(solutionDescription, solutionDescription.value.trim().length >= 20);
    } else {
        solutionDescription.style.borderColor = "";
    }
});

// Form submission
form.addEventListener('submit', (e) => {
    e.preventDefault();
    const validations = validateForm();

    // Highlight invalid fields
    setBorderColor(fullNameInput, validations["full-name"]);
    setBorderColor(emailInput, validations["email"]);
    setBorderColor(orderNoInput, validations["order-no"]);
    setBorderColor(productCodeInput, validations["product-code"]);
    setBorderColor(quantityInput, validations["quantity"]);
    setGroupBorderColor(complaintsGroup, validations["complaints-group"]);
    setBorderColor(complaintDescription, validations["complaint-description"]);
    setGroupBorderColor(solutionsGroup, validations["solutions-group"]);
    setBorderColor(solutionDescription, validations["solution-description"]);

    // Show message if all valid
    if (isValid(validations)) {
        messageBox.innerText = "Form submitted successfully!";
        messageBox.style.color = "green";
    } else {
        messageBox.innerText = "Please fix the errors highlighted in red.";
        messageBox.style.color = "red";
    }
});
